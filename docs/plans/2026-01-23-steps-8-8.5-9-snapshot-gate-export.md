fn create_snapshot_with_metadata(
    workspace_path: String,
    snapshot_meta: SnapshotMetadata,
) -> Result<String, String> {
    let snapshot_id = snapshot_meta.id.clone();
    let snapshot_dir = Path::new(&workspace_path)
        .join(".epris/history")
        .join(&snapshot_id);

    // Create snapshot directory
    std::fs::create_dir_all(&snapshot_dir)
        .map_err(|e| format!("Failed to create snapshot dir: {}", e))?;

    // Copy whitelisted directories
    for dir_name in SNAPSHOT_WHITELIST {
        let src_dir = Path::new(&workspace_path).join(dir_name);
        if src_dir.exists() {
            let dest_dir = snapshot_dir.join(dir_name);
            if let Err(e) = fs_extra::dir::copy(
                &src_dir,
                &snapshot_dir,
                &fs_extra::dir::CopyOptions::new()
            ) {
                return Err(format!("Failed to copy {}: {}", dir_name, e));
            }
        }
    }

    // Write metadata
    let meta_path = snapshot_dir.join("meta.json");
    let meta_json = serde_json::to_string_pretty(&snapshot_meta)
        .map_err(|e| format!("Failed to serialize metadata: {}", e))?;
    std::fs::write(&meta_path, meta_json)
        .map_err(|e| format!("Failed to write metadata: {}", e))?;

    Ok(snapshot_id)
}
```

fn restore_snapshot(workspace_path: &str, snapshot_id: &str) -> Result<(), String> {
    let snapshot_dir = Path::new(workspace_path)
        .join(".epris/history")
        .join(snapshot_id);

    if !snapshot_dir.exists() {
        return Err(format!("Snapshot {} not found", snapshot_id));
    }

    // Remove current content (whitelisted directories only)
    for dir_name in SNAPSHOT_WHITELIST {
        let target_dir = Path::new(workspace_path).join(dir_name);
        if target_dir.exists() {
            std::fs::remove_dir_all(&target_dir)
                .map_err(|e| format!("Failed to remove {}: {}", dir_name, e))?;
        }
    }

    // Copy snapshot content back
    for dir_name in SNAPSHOT_WHITELIST {
        let src_dir = snapshot_dir.join(dir_name);
        if src_dir.exists() {
            let dest_dir = Path::new(workspace_path).join(dir_name);
            fs_extra::dir::copy(
                &src_dir,
                Path::new(workspace_path),
                &fs_extra::dir::CopyOptions::new()
            ).map_err(|e| format!("Failed to restore {}: {}", dir_name, e))?;
        }
    }

    Ok(())
}
```

```rust
fn check_forbidden_files_modified(
    workspace_path: &str,
    snapshot_id: &str,
) -> Result<Vec<String>, String> {
    let snapshot_dir = Path::new(workspace_path)
        .join(".epris/history")
        .join(snapshot_id);

    let mut modified_forbidden = Vec::new();

    for forbidden_file in FORBIDDEN_FILES {
        let workspace_file = Path::new(workspace_path).join(forbidden_file);
        let snapshot_file = snapshot_dir.join(forbidden_file);

        if workspace_file.exists() && snapshot_file.exists() {
            let workspace_content = std::fs::read(&workspace_file)
                .map_err(|e| format!("Failed to read {}: {}", forbidden_file, e))?;
            let snapshot_content = std::fs::read(&snapshot_file)
                .map_err(|e| format!("Failed to read snapshot {}: {}", forbidden_file, e))?;

            if workspace_content != snapshot_content {
                modified_forbidden.push(forbidden_file.to_string());
            }
        } else if workspace_file.exists() != snapshot_file.exists() {
            modified_forbidden.push(forbidden_file.to_string());
        }
    }

    Ok(modified_forbidden)
}
```

fn log_security_violation(
    workspace_path: &str,
    forbidden_files: &[String],
    prompt: &str,
) -> Result<(), String> {
    let log_dir = Path::new(workspace_path).join("logs");
    std::fs::create_dir_all(&log_dir)
        .map_err(|e| format!("Failed to create logs dir: {}", e))?;

    let log_path = log_dir.join("security.log");
    let log_entry = format!(
        "[{}] SECURITY VIOLATION\nPrompt: {}\nModified forbidden files: {}\n\n",
        chrono::Utc::now().to_rfc3339(),
        prompt,
        forbidden_files.join(", ")
    );

    std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(&log_path)
        .and_then(|mut file| std::io::Write::write_all(&mut file, log_entry.as_bytes()))
        .map_err(|e| format!("Failed to write security log: {}", e))
}

async fn send_prompt(
    state: tauri::State<'_, Mutex<OpenCodeState>>,
    workspace_path: String,
    prompt: String,
) -> Result<PromptResponse, String> {
    // Step 1: Auto-save snapshot before modification
    let parent_id = get_current_head(workspace_path.clone())?;
    let session_id = {
        let st = state.lock().map_err(|e| e.to_string())?;
        st.session_id.clone().unwrap_or_else(|| "none".to_string())
    };

    let snapshot_id = auto_save_snapshot(
        workspace_path.clone(),
        prompt.clone(),
        parent_id,
        session_id.clone(),
    )?;

    // Step 2: Send prompt to OpenCode
    let port = {
        let st = state.lock().map_err(|e| e.to_string())?;
        st.port.ok_or("OpenCode not running")?
    };

    let session_response = if let Ok(st) = state.lock() {
        match &st.session_id {
            Some(sid) => {
                let client = reqwest::Client::new();
                client
                    .post(format!("http://127.0.0.1:{}/session/{}/message", port, sid))
                    .json(&serde_json::json!({
                        "content": format!("{}\n\n{}", REMOTION_INSTRUCTION, prompt)
                    }))
                    .send()
                    .await
                    .map_err(|e| e.to_string())?;
                sid.clone()
            }
            None => {
                let client = reqwest::Client::new();
                let res = client
                    .post(format!("http://127.0.0.1:{}/session", port))
                    .json(&serde_json::json!({
                        "workspaceFolder": workspace_path,
                        "message": format!("{}\n\n{}", REMOTION_INSTRUCTION, prompt)
                    }))
                    .send()
                    .await
                    .map_err(|e| e.to_string())?;

                let json: serde_json::Value = res.json().await.map_err(|e| e.to_string())?;
                let new_sid = json["sessionId"]
                    .as_str()
                    .ok_or("No sessionId in response")?
                    .to_string();

                if let Ok(mut st) = state.lock() {
                    st.session_id = Some(new_sid.clone());
                }
                new_sid
            }
        }
    } else {
        return Err("Failed to lock OpenCode state".to_string());
    };

    // Wait for OpenCode to finish
    std::thread::sleep(std::time::Duration::from_secs(3));

    // Step 2.5: Security check for forbidden files
    let forbidden_files = check_forbidden_files_modified(&workspace_path, &snapshot_id)?;

    if !forbidden_files.is_empty() {
        // Rollback to snapshot
        checkout_snapshot(workspace_path.clone(), snapshot_id.clone())?;

        // Log violation
        log_security_violation(&workspace_path, &forbidden_files, &prompt)?;

        return Ok(PromptResponse {
            success: false,
            message: format!(
                "🔒 Security: AI attempted to modify protected files: {}. Changes reverted.",
                forbidden_files.join(", ")
            ),
            gate_result: None,
            snapshot_id: Some(snapshot_id),
        });
    }

    // Step 3: Run gate validation
    let gate_result = gate_loop(workspace_path.clone(), session_response, 2, port).await?;

    // Step 4: Update snapshot with gate result
    update_snapshot_gate_result(workspace_path.clone(), snapshot_id.clone(), gate_result.clone())?;

    // Step 5: Update HEAD
    set_head(workspace_path, snapshot_id.clone())?;

    Ok(PromptResponse {
        success: gate_result.passed,
        message: if gate_result.passed {
            "Gate passed".to_string()
        } else {
            format!("Gate failed: {}", gate_result.error.unwrap_or_default())
        },
        gate_result: Some(gate_result),
        snapshot_id: Some(snapshot_id),
    })
}

#[tauri::command]
fn get_snapshot_dag(workspace_path: String) -> Result<SnapshotDAG, String> {
    let history_dir = Path::new(&workspace_path).join(".epris/history");

    if !history_dir.exists() {
        return Ok(SnapshotDAG {
            nodes: Vec::new(),
            edges: Vec::new(),
        });
    }

    let mut nodes = Vec::new();
    let mut edges = Vec::new();

    for entry in std::fs::read_dir(history_dir).map_err(|e| e.to_string())? {
        let entry = entry.map_err(|e| e.to_string())?;
        let meta_path = entry.path().join("meta.json");

        if meta_path.exists() {
            let meta_content = std::fs::read_to_string(&meta_path)
                .map_err(|e| format!("Failed to read metadata: {}", e))?;

            let meta: SnapshotMetadata = serde_json::from_str(&meta_content)
                .map_err(|e| format!("Failed to parse metadata: {}", e))?;

            if let Some(parent_id) = &meta.parent_id {
                edges.push(Edge {
                    from: parent_id.clone(),
                    to: meta.id.clone(),
                });
            }

            nodes.push(meta);
        }
    }

    Ok(SnapshotDAG { nodes, edges })
}
```

async fn export_video(

    let result = (|| -> Result<String, String> {
        let output_path = Path::new(&workspace_path).join("out/video.mp4");

        // Ensure out directory exists
        let out_dir = Path::new(&workspace_path).join("out");

            format!("Failed to execute export command: {}", e)

            // Log error

import { useState, useEffect, useCallback } from "react";
import { invoke } from "@tauri-apps/api/core";

export interface SnapshotMetadata {
  id: string;
  name: string;
  description: string;
  timestamp: string;
  parent_id: string | null;
  is_manual: boolean;
  prompt: string | null;
  session_id: string;
  gate_result: any | null;
}

export interface SnapshotDAG {
  nodes: SnapshotMetadata[];
  edges: Array<{ from: string; to: string }>;
}

export function useSnapshots(workspacePath: string | undefined) {
  const [snapshots, setSnapshots] = useState<SnapshotMetadata[]>([]);
  const [currentHead, setCurrentHead] = useState<string>("root");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const refresh = useCallback(async () => {
    if (!workspacePath) return;

    setLoading(true);
    setError(null);

    try {
      const [snapshotList, head] = await Promise.all([
        invoke<SnapshotMetadata[]>("list_snapshots", { workspacePath }),
        invoke<string>("get_current_head", { workspacePath }),
      ]);

      setSnapshots(snapshotList);
      setCurrentHead(head);
    } catch (err) {
      const errorMsg = err instanceof Error ? err.message : String(err);
      setError(errorMsg);
    } finally {
      setLoading(false);
    }
  }, [workspacePath]);

  useEffect(() => {
    refresh();
  }, [refresh]);

  const checkout = useCallback(
    async (snapshotId: string) => {
      if (!workspacePath) throw new Error("No workspace path");

      await invoke("checkout_snapshot", {
        workspacePath,
        snapshotId,
      });

      setCurrentHead(snapshotId);
    },
    [workspacePath],
  );

  const updateMetadata = useCallback(
    async (snapshotId: string, name: string, description: string) => {
      if (!workspacePath) throw new Error("No workspace path");

      await invoke("update_snapshot_metadata", {
        workspacePath,
        snapshotId,
        name,
        description,
      });

      await refresh();
    },
    [workspacePath, refresh],
  );

  return {
    snapshots,
    currentHead,
    loading,
    error,
    refresh,
    checkout,
    updateMetadata,
  };
}

import { useCallback } from "react";
import { invoke } from "@tauri-apps/api/core";

export function useRollback(
  workspacePath: string | undefined,
  onSuccess?: () => void,
) {
  const rollback = useCallback(async () => {
    if (!workspacePath) {
      throw new Error("No workspace path");
    }

    try {
      // Get current HEAD
      const currentHead = await invoke<string>("get_current_head", {
        workspacePath,
      });

      if (currentHead === "root") {
        throw new Error("No snapshot to rollback to");
      }

      // Get all snapshots to find parent
      const snapshots = await invoke<
        Array<{
          id: string;
          parent_id: string | null;
        }>
      >("list_snapshots", { workspacePath });

      const current = snapshots.find((s) => s.id === currentHead);
      if (!current || !current.parent_id) {
        throw new Error("No parent snapshot found");
      }

      // Checkout parent snapshot
      await invoke("checkout_snapshot", {
        workspacePath,
        snapshotId: current.parent_id,
      });

      onSuccess?.();
    } catch (err) {
      throw err;
    }
  }, [workspacePath, onSuccess]);

  return { rollback };
}
```

# Snapshot System

## Overview

The snapshot system provides Git-like version control for workspace modifications, with automatic snapshots before each AI prompt and manual save capabilities.

## Architecture

### Storage Structure
```

workspace/
.epris/
HEAD # Current snapshot ID
history/
root/ # Initial snapshot
meta.json
src/
public/
<uuid-1>/ # Auto snapshot
meta.json
src/
public/
<uuid-2>/ # Manual snapshot
...

````

### Snapshot Metadata

Each snapshot contains:
- `id`: Unique UUID
- `name`: Human-readable name (auto-generated or user-provided)
- `description`: Optional description
- `timestamp`: ISO 8601 timestamp
- `parent_id`: Parent snapshot ID (for DAG)
- `is_manual`: Boolean flag
- `prompt`: Original prompt (for auto snapshots)
- `session_id`: OpenCode session ID
- `gate_result`: Validation results

## Security

### Forbidden Files

The following files cannot be modified by AI:
- `package.json`
- `pnpm-lock.yaml`
- `tsconfig.json`
- `vite.config.ts`
- `index.html`

Any attempt to modify these files triggers:
1. Automatic rollback to previous snapshot
2. Security log entry in `logs/security.log`
3. Error message to user

### Whitelisted Content

Only these directories are versioned:
- `src/` - Source code
- `public/` - Static assets

## Commands

### Rust Commands

- `get_current_head(workspace_path)` - Get current HEAD snapshot ID
- `set_head(workspace_path, snapshot_id)` - Update HEAD pointer
- `auto_save_snapshot(...)` - Create auto snapshot before prompt
- `manual_save_snapshot(...)` - Create manual snapshot
- `checkout_snapshot(workspace_path, snapshot_id)` - Restore snapshot
- `list_snapshots(workspace_path)` - List all snapshots
- `get_snapshot_dag(workspace_path)` - Get full DAG structure
- `update_snapshot_metadata(...)` - Update snapshot name/description

### Frontend Hooks

- `useSnapshots(workspacePath)` - Snapshot list and management
- `useManualSave(workspacePath)` - Manual save functionality
- `useRollback(workspacePath, onSuccess)` - Rollback to parent
